# ddl-auto: create 라서 실행할 때마다 테이블을 지우고 다시 만든다.

Hibernate: 
    drop table if exists tbl_member
Hibernate: 
    create table tbl_member (
        member_no integer not null auto_increment,
        enroll_date datetime(6),
        address varchar(900),
        member_id varchar(10) not null,
        member_name varchar(255),
        member_pwd varchar(255) not null,
        phone varchar(255),
        status char(1) default 'Y',
        member_role enum ('ROLE_ADMIN','ROLE_MEMBER'),
        primary key (member_no)
    ) engine=InnoDB
Hibernate: 
    alter table tbl_member 
       add constraint UK… unique (member_id)
Hibernate: 
    alter table tbl_member 
       add constraint UK… unique (phone)
